38  class StreamPollFeederTest {
39  
40      /** An input that never has data and never ends, like an idle terminal. */
41      private static final class IdleInputStream extends InputStream {
42          private final CountDownLatch never = new CountDownLatch(1);
43  
44          @Override
45          public int available() {
46              return 0;
47          }
48  
49          @Override
50          public int read() throws IOException {
51              try {
52                  never.await();
53              } catch (InterruptedException e) {
54                  throw new IOException(e);
55              }
56              return -1;
57          }
58      }
59  
60      private static final class CountingOutputStream extends ByteArrayOutputStream {
61          private volatile boolean closed;
62  
63          @Override
64          public void close() {
65              closed = true;
66          }
67      }
68  
69      @Test
70      void waitUntilDoneWithIdleInput() {
71          CountingOutputStream out = new CountingOutputStream();
72          StreamPollFeeder feeder = new StreamPollFeeder(new IdleInputStream(), out);
73  
74          assertTimeoutPreemptively(Duration.ofSeconds(5), () -> {
75              feeder.start();
76              Thread.sleep(150);
77              feeder.waitUntilDone();
78          });
79  
80          assertNull(feeder.getException());
81          assertTrue(feeder.isDone());
82          assertTrue(out.closed, "output must be closed when feeding ends");
83      }
84  
85      @Test
86      void waitUntilDoneWithContinuousData() {
87          InputStream continuous = new InputStream() {
88              @Override
89              public int available() {
90                  return 1;
91              }
92  
93              @Override
94              public int read() {
95                  return 0;
96              }
97  
98              @Override
99              public int read(byte[] b, int off, int len) {
100                 return 1;
101             }
102         };
103         StreamPollFeeder feeder = new StreamPollFeeder(continuous, new ByteArrayOutputStream() {
104             @Override
105             public synchronized void write(byte[] b, int off, int len) {
106                 // discard
107             }
108         });
109 
110         assertTimeoutPreemptively(Duration.ofSeconds(5), () -> {
111             feeder.start();
112             feeder.waitUntilDone();
113         });
114         assertNull(feeder.getException());
115     }
116 
117     @Test
118     void dataIsCopied() throws Exception {
119         StringBuilder data = new StringBuilder();
120         for (int i = 0; i < 100; i++) {
121             data.append("TestData");
122         }
123         ByteArrayOutputStream out = new ByteArrayOutputStream();
124         StreamPollFeeder feeder =
125                 new StreamPollFeeder(new ByteArrayInputStream(data.toString().getBytes(StandardCharsets.UTF_8)), out);
126 
127         assertTimeoutPreemptively(Duration.ofSeconds(10), () -> {
128             feeder.start();
129             while (out.size() < data.length()) {
130                 Thread.sleep(20);
131             }
132             feeder.waitUntilDone();
133         });
134 
135         assertNull(feeder.getException());
136         assertEquals(data.toString(), out.toString("UTF-8"));
137     }

139     @Test
140     @DisabledOnOs(OS.WINDOWS)
141     void executeCommandLineDoesNotWaitForIdleSystemIn() throws Exception {
142         InputStream originalIn = System.in;
143         System.setIn(new IdleInputStream());
144         try {
145             Commandline cli = new Commandline();
146             cli.setExecutable(javaExecutable().getAbsolutePath());
147             cli.createArg().setValue("-version");
148 
149             int exitCode = assertTimeoutPreemptively(
150                     Duration.ofSeconds(30),
151                     () -> CommandLineUtils.executeCommandLine(
152                             cli,
153                             System.in,
154                             new CommandLineUtils.StringStreamConsumer(),
155                             new CommandLineUtils.StringStreamConsumer()));
156             assertEquals(0, exitCode);
157         } finally {
158             System.setIn(originalIn);
159         }
160     }
161 
162     @Test
163     @DisabledOnOs(OS.WINDOWS)
164     void executeCommandLineStillDeliversEndOfInputToFiniteStream() throws Exception {
165         Commandline cli = new Commandline();
166         cli.setExecutable("cat");
167         CommandLineUtils.StringStreamConsumer out = new CommandLineUtils.StringStreamConsumer();
168 
169         int exitCode = assertTimeoutPreemptively(
170                 Duration.ofSeconds(30),
171                 () -> CommandLineUtils.executeCommandLine(
172                         cli,
173                         new ByteArrayInputStream("hello".getBytes(StandardCharsets.UTF_8)),
174                         out,
175                         new CommandLineUtils.StringStreamConsumer()));
176 
177         assertEquals(0, exitCode);
178         assertEquals("hello", out.getOutput().trim());
179     }
180 
181     private static File javaExecutable() {
182         File java =
183                 new File(System.getProperty("java.home"), Os.isFamily(Os.FAMILY_WINDOWS) ? "bin/java.exe" : "bin/java");
184         assertTrue(java.exists(), java + " doesn't exist");
185         return java;
186     }
